-- Step 2: Create Sample Tables with Data

-- Create Customers table
CREATE TABLE dbo.Customers
(
    CustomerID INT PRIMARY KEY IDENTITY(1,1),
    FirstName NVARCHAR(100) NOT NULL,
    LastName NVARCHAR(100) NOT NULL,
    Email NVARCHAR(100) NOT NULL UNIQUE,
    Phone NVARCHAR(20),
    City NVARCHAR(100),
    State NVARCHAR(50),
    Country NVARCHAR(100),
    CreatedDate DATETIME2 DEFAULT SYSDATETIME(),
    ModifiedDate DATETIME2 DEFAULT SYSDATETIME()
);

-- Create Orders table
CREATE TABLE dbo.Orders
(
    OrderID INT PRIMARY KEY IDENTITY(1,1),
    CustomerID INT NOT NULL FOREIGN KEY REFERENCES dbo.Customers(CustomerID),
    OrderDate DATETIME2 NOT NULL,
    ShipDate DATETIME2,
    TotalAmount DECIMAL(10,2) NOT NULL,
    Status NVARCHAR(50) DEFAULT 'Pending',
    CreatedDate DATETIME2 DEFAULT SYSDATETIME()
);

-- Create OrderDetails table
CREATE TABLE dbo.OrderDetails
(
    OrderDetailID INT PRIMARY KEY IDENTITY(1,1),
    OrderID INT NOT NULL FOREIGN KEY REFERENCES dbo.Orders(OrderID),
    ProductID INT NOT NULL,
    Quantity INT NOT NULL,
    UnitPrice DECIMAL(10,2) NOT NULL,
    LineTotal AS (Quantity * UnitPrice) PERSISTED
);

-- Create Products table
CREATE TABLE dbo.Products
(
    ProductID INT PRIMARY KEY IDENTITY(1,1),
    ProductName NVARCHAR(200) NOT NULL,
    Category NVARCHAR(100),
    ListPrice DECIMAL(10,2),
    StockQuantity INT DEFAULT 0
);
GO

-- Step 3: Populate Sample Data

-- Populate Customers
SET IDENTITY_INSERT dbo.Customers ON;

DECLARE @i INT = 1;
WHILE @i <= 10000
BEGIN
    INSERT INTO dbo.Customers (CustomerID, FirstName, LastName, Email, Phone, City, State, Country)
    VALUES 
    (
        @i,
        'Customer' + CAST(@i AS NVARCHAR(10)),
        'Test' + CAST(@i % 100 AS NVARCHAR(10)),
        'customer' + CAST(@i AS NVARCHAR(10)) + '@example.com',
        '555-' + FORMAT(@i, '0000'),
        'City' + CAST(@i % 50 AS NVARCHAR(10)),
        'ST',
        'USA'
    );
    SET @i = @i + 1;
END

SET IDENTITY_INSERT dbo.Customers OFF;
GO

-- Populate Products
INSERT INTO dbo.Products (ProductName, Category, ListPrice, StockQuantity)
SELECT 
    'Product ' + CAST(ROW_NUMBER() OVER (ORDER BY @@SPID) AS NVARCHAR(10)),
    'Category' + CAST(ROW_NUMBER() OVER (ORDER BY @@SPID) % 10 AS NVARCHAR(10)),
    ROUND(RAND() * 1000 + 10, 2),
    CAST(RAND() * 1000 AS INT)
FROM sys.objects
WHERE ROW_NUMBER() OVER (ORDER BY @@SPID) <= 100;
GO

-- Populate Orders and OrderDetails
DECLARE @CustomerID INT, @OrderID INT;
DECLARE @OrderCount INT = 0;

DECLARE @CustomerCursor CURSOR;
SET @CustomerCursor = CURSOR FOR SELECT CustomerID FROM dbo.Customers ORDER BY CustomerID;
OPEN @CustomerCursor;

FETCH NEXT FROM @CustomerCursor INTO @CustomerID;
WHILE @@FETCH_STATUS = 0
BEGIN
    DECLARE @OrdersPerCustomer INT = CAST(RAND() * 5 + 1 AS INT);
    DECLARE @OrderIdx INT = 0;
    
    WHILE @OrderIdx < @OrdersPerCustomer
    BEGIN
        INSERT INTO dbo.Orders (CustomerID, OrderDate, TotalAmount, Status)
        VALUES 
        (
            @CustomerID,
            DATEADD(DAY, -CAST(RAND() * 365 AS INT), GETDATE()),
            ROUND(RAND() * 5000 + 100, 2),
            CASE CAST(RAND() * 3 AS INT) WHEN 0 THEN 'Pending' WHEN 1 THEN 'Shipped' ELSE 'Delivered' END
        );
        
        SET @OrderID = SCOPE_IDENTITY();
        
        -- Add order details
        DECLARE @DetailCount INT = CAST(RAND() * 5 + 1 AS INT);
        DECLARE @DetailIdx INT = 0;
        
        WHILE @DetailIdx < @DetailCount
        BEGIN
            INSERT INTO dbo.OrderDetails (OrderID, ProductID, Quantity, UnitPrice)
            VALUES
            (
                @OrderID,
                CAST(RAND() * 100 + 1 AS INT),
                CAST(RAND() * 10 + 1 AS INT),
                ROUND(RAND() * 500 + 10, 2)
            );
            
            SET @DetailIdx = @DetailIdx + 1;
        END
        
        SET @OrderIdx = @OrderIdx + 1;
    END
    
    FETCH NEXT FROM @CustomerCursor INTO @CustomerID;
END

CLOSE @CustomerCursor;
DEALLOCATE @CustomerCursor;
GO

Query Performance Baseline Analysis
-- Server Performance Baseline DMV
-- Returns top wait statistics affecting performance
USE PerformanceLab;
GO

SELECT TOP 20
    wait_type,
    waiting_tasks_count,
    wait_time_ms,
    signal_wait_time_ms,
    avg_wait_time_ms = wait_time_ms / waiting_tasks_count,
    pct_wait = CONVERT(DECIMAL(5,2), 100.0 * wait_time_ms / 
              (SELECT SUM(wait_time_ms) FROM sys.dm_os_waiting_tasks)) 
FROM sys.dm_os_waiting_tasks
WHERE wait_duration_ms > 0
ORDER BY wait_time_ms DESC;
GO
--------------------------------------------

-- Identify CPU-consuming queries currently running
SELECT
    session_id,
    request_id,
    start_time,
    status,
    command,
    cpu_time,
    total_elapsed_time,
    reads,
    writes,
    logical_reads = (reads + logical_reads),
    sql_text = SUBSTRING(st.text, (qs.statement_start_offset / 2) + 1, 
                         ((CASE WHEN qs.statement_end_offset = -1 
                           THEN LEN(CONVERT(NVARCHAR(MAX), st.text)) * 2 
                           ELSE qs.statement_end_offset 
                          END - qs.statement_start_offset) / 2) + 1)
FROM sys.dm_exec_requests qs
CROSS APPLY sys.dm_exec_sql_text(qs.sql_handle) st
WHERE session_id > 50
    AND status NOT IN ('background', 'sleeping')
ORDER BY cpu_time DESC;
GO
-----------------------------------------------

-- Find most expensive queries from cache
-- Note: requires data to have been accumulated
SELECT TOP 20
    total_worker_time / 1000000 AS total_cpu_sec,
    execution_count,
    (total_worker_time / 1000000) / execution_count AS avg_cpu_sec,
    total_logical_reads,
    total_logical_reads / execution_count AS avg_logical_reads,
    total_elapsed_time / 1000000 AS total_elapsed_sec,
    SUBSTRING(qt.text, 1, 100) AS query_text
FROM sys.dm_exec_query_stats qs
CROSS APPLY sys.dm_exec_sql_text(qs.sql_handle) qt
WHERE qt.text NOT LIKE '%dm_exec%'
ORDER BY qs.total_worker_time DESC;
GO

-- Identify I/O bottlenecks
SELECT
    database_id,
    file_id,
    filename = mf.name,
    num_of_reads,
    num_of_writes,
    io_stall_read_ms,
    io_stall_write_ms,
    io_stall_total_ms = io_stall_read_ms + io_stall_write_ms,
    avg_read_stall_ms = 
        CASE WHEN num_of_reads = 0 THEN 0 
             ELSE io_stall_read_ms / num_of_reads 
        END,
    avg_write_stall_ms = 
        CASE WHEN num_of_writes = 0 THEN 0 
             ELSE io_stall_write_ms / num_of_writes 
        END
FROM sys.dm_io_virtual_file_stats(NULL, NULL) AS dvfs
INNER JOIN sys.master_files AS mf ON dvfs.database_id = mf.database_id
    AND dvfs.file_id = mf.file_id
WHERE database_id > 4  -- Exclude system databases
ORDER BY io_stall_total_ms DESC;
GO

-- Check current memory consumption
SELECT
    counter_name,
    cntr_value,
    cntr_type,
    CASE 
        WHEN counter_name = 'Memory Grants Outstanding' THEN cntr_value
        WHEN counter_name = 'Memory Grants Pending' THEN cntr_value
        ELSE NULL 
    END AS critical_memory_metric
FROM sys.dm_os_performance_counters
WHERE counter_name IN 
    ('Memory Grants Outstanding', 'Memory Grants Pending', 
     'Total Server Memory (KB)', 'Target Server Memory (KB)',
     'Pages/sec', 'Page Faults/sec')
    AND object_name LIKE '%Memory Manager%'
ORDER BY counter_name;
GO
