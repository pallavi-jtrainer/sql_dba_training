# Check Windows Version
Write-Host "`nWindows Version Check:" -ForegroundColor Yellow
$osVersion = [System.Environment]::OSVersion.VersionString
Write-Host $osVersion

# Get detailed Windows version
$winVersion = Get-ItemProperty 'HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion'
Write-Host "Build: $($winVersion.CurrentBuild)"

# Check RAM
Write-Host "`nMemory Check:" -ForegroundColor Yellow
$totalMemory = (Get-WmiObject -Class Win32_ComputerSystem).TotalPhysicalMemory / 1GB
Write-Host "Total RAM: $([Math]::Round($totalMemory, 2)) GB"
Write-Host "Minimum Required: 16 GB"
Write-Host "Recommended: 32 GB"

# Check Disk Space on C: drive
Write-Host "`nDisk Space Check:" -ForegroundColor Yellow
$cDrive = Get-Volume -DriveLetter C
$freeSpace = $cDrive.SizeRemaining / 1GB
Write-Host "Free Space on C: $([Math]::Round($freeSpace, 2)) GB"
Write-Host "Minimum Required: 50 GB"

# Check PowerShell Version
Write-Host "`nPowerShell Version Check:" -ForegroundColor Yellow
$PSVersion = $PSVersionTable.PSVersion
Write-Host "Current Version: $($PSVersion.Major).$($PSVersion.Minor)"
Write-Host "Minimum Required: 5.1"

# Check .NET Framework
Write-Host "`n.NET Framework Check:" -ForegroundColor Yellow
$dotnetKey = Get-ItemProperty 'HKLM:\SOFTWARE\Microsoft\NET Framework Setup\NDP\v4\Full'
if ($dotnetKey.Release -ge 528040) {
    Write-Host ".NET Framework 4.8 or later installed (Release: $($dotnetKey.Release))"
} else {
    Write-Host ".NET Framework needs to be updated" -ForegroundColor Red
}

# Summary
Write-Host "`n=== Summary ===" -ForegroundColor Green
if ($totalMemory -ge 16 -and $freeSpace -ge 50) {
    Write-Host "System is READY for SQL Server 2025 installation" -ForegroundColor Green
} else {
    Write-Host "System needs adjustment before installation" -ForegroundColor Red
}

# PowerShell Script: Create SQL Server Service Account
# Run as Administrator

param(
    [string]$AccountName = "SQLServiceAccount",
    [string]$Password = "P@ssw0rd123!"
)

Write-Host "Creating SQL Server Service Account..." -ForegroundColor Green

# Check if account already exists
$userExists = Get-LocalUser -Name $AccountName -ErrorAction SilentlyContinue

if ($null -eq $userExists) {
    # Create secure password
    $securePassword = ConvertTo-SecureString -AsPlainText -Force -String $Password
    
    # Create local user
    New-LocalUser -Name $AccountName `
                  -Password $securePassword `
                  -FullName "SQL Server Service Account" `
                  -Description "Service account for SQL Server 2025" `
                  -PasswordNeverExpires
    
    Write-Host "User $AccountName created successfully" -ForegroundColor Green
    
    # Add to Administrators group (for initial setup)
    Add-LocalGroupMember -Group "Administrators" -Member $AccountName
    Write-Host "User added to Administrators group" -ForegroundColor Green
} else {
    Write-Host "User $AccountName already exists" -ForegroundColor Yellow
}

# Verify user exists
$user = Get-LocalUser -Name $AccountName
Write-Host "Account Details:" -ForegroundColor Cyan
Write-Host "Name: $($user.Name)"
Write-Host "Description: $($user.Description)"
Write-Host "Enabled: $($user.Enabled)"

# PowerShell: Check SQL Server Services Status
# Run after installation

Write-Host "=== SQL Server Services Status ===" -ForegroundColor Green

# Get all SQL Server services
$sqlServices = Get-Service -Name "MSSQL*","SQLSERVERAGENT*","SQLBROWSER*" -ErrorAction SilentlyContinue

if ($sqlServices.Count -gt 0) {
    foreach ($service in $sqlServices) {
        $status = if ($service.Status -eq "Running") { "✓ Running" } else { "✗ Stopped" }
        Write-Host "$($service.DisplayName): $status" -ForegroundColor Cyan
    }
} else {
    Write-Host "No SQL Server services found. Installation may not be complete." -ForegroundColor Red
}

# Start SQL Server Agent if stopped
$agentService = Get-Service -Name "SQLSERVERAGENT" -ErrorAction SilentlyContinue
if ($agentService -and $agentService.Status -ne "Running") {
    Write-Host "`nStarting SQL Server Agent..." -ForegroundColor Yellow
    Start-Service -Name "SQLSERVERAGENT"
    Start-Sleep -Seconds 3
    $agentService.Refresh()
    Write-Host "SQL Server Agent is now $($agentService.Status)" -ForegroundColor Green
}

-- SQL Script: Check Current Memory Configuration

-- View 1: Current memory settings
SELECT 
    'max server memory' AS [Configuration],
    value_in_use AS [Current Value (MB)],
    CAST(value_in_use / 1024 AS NUMERIC(10, 2)) AS [In GB],
    value AS [Config Value (MB)]
FROM sys.configurations
WHERE name = 'max server memory (MB)'
UNION ALL
SELECT 
    'min server memory',
    value_in_use,
    CAST(value_in_use / 1024 AS NUMERIC(10, 2)),
    value
FROM sys.configurations
WHERE name = 'min server memory (MB)';

-- View 2: Actual memory usage
SELECT 
    'Physical Memory (MB)' AS [Type],
    (SELECT cntr_value / 1024.0 FROM sys.dm_os_performance_counters 
     WHERE counter_name = 'Total Server Memory (KB)') AS [Current (MB)],
    (SELECT cntr_value / 1024.0 FROM sys.dm_os_performance_counters 
     WHERE counter_name = 'Target Server Memory (KB)') AS [Target (MB)]
UNION ALL
SELECT 
    'Free Memory (MB)',
    (SELECT cntr_value / 1024.0 FROM sys.dm_os_performance_counters 
     WHERE counter_name = 'Free Memory (KB)'),
    NULL
UNION ALL
SELECT 
    'Stolen Server Memory (MB)',
    (SELECT cntr_value / 1024.0 FROM sys.dm_os_performance_counters 
     WHERE counter_name = 'Stolen Server Memory (KB)'),
    NULL;

-- View 3: Memory breakdown by component
SELECT 
    CASE type 
        WHEN 'MEMORYCLERK_SQLBUFFERPOOL' THEN 'Buffer Pool'
        WHEN 'MEMORYCLERK_SQLOPTIMIZER' THEN 'Optimizer'
        WHEN 'MEMORYCLERK_SQLEXECUTION' THEN 'Execution'
        ELSE type 
    END AS [Memory Component],
    SUM(pages_kb) / 1024.0 AS [Size (MB)],
    SUM(pages_in_use_kb) / 1024.0 AS [In Use (MB)],
    SUM(pages_free_kb) / 1024.0 AS [Free (MB)]
FROM sys.dm_os_memory_clerks
GROUP BY type
ORDER BY SUM(pages_kb) DESC;

CREATE ROLE db_developers;
CREATE ROLE db_analysts;
CREATE ROLE db_readonly;

GRANT SELECT, INSERT, UPDATE, DELETE ON dbo.Transactions TO db_developers;
GRANT SELECT, INSERT, UPDATE, DELETE ON dbo.AnalyticsData TO db_developers;

-- db_analysts: SELECT only
GRANT SELECT ON dbo.Users TO db_analysts;
GRANT SELECT ON dbo.Transactions TO db_analysts;
GRANT SELECT ON dbo.AnalyticsData TO db_analysts;

-- db_readonly: SELECT only on specific tables
GRANT SELECT ON dbo.AnalyticsData TO db_readonly;

-- Add users to roles
ALTER ROLE db_developers ADD MEMBER [AppUser1];
ALTER ROLE db_analysts ADD MEMBER [AppUser2];
ALTER ROLE db_readonly ADD MEMBER [DBAUser];

-- Verify role membership
SELECT 
    dp.name AS [Role Name],
    m.name AS [Member Name],
    m.type_desc
FROM sys.database_role_members drm
JOIN sys.database_principals dp ON drm.role_principal_id = dp.principal_id
JOIN sys.database_principals m ON drm.member_principal_id = m.principal_id
WHERE dp.type_desc = 'DATABASE_ROLE'
ORDER BY dp.name;

FROM msdb.dbo.backupset
WHERE database_name = 'SampleDB'

TO DISK = N'C:\Backups\SampleDB_Log_20250124_1000.trn'
WITH 
    COMPRESSION,
    INIT,
    NAME = N'SampleDB Transaction Log Backup 1',
    DESCRIPTION = N'Transaction log backup capturing all changes since last backup',
    STATS = 10;

-- Wait a bit and insert more data
WAITFOR DELAY '00:00:05';

USE SampleDB;
INSERT INTO dbo.Transactions (UserID, Amount, Status) VALUES
(1, 500.00, 'Completed'),
(2, 750.00, 'Pending');

-- Create another transaction log backup
BACKUP LOG SampleDB
TO DISK = N'C:\Backups\SampleDB_Log_20250124_1001.trn'
WITH 
    COMPRESSION,
    INIT,
    NAME = N'SampleDB Transaction Log Backup 2',
    STATS = 10;

-- Query log backup history
SELECT 
    database_name,
    backup_start_date,
    backup_finish_date,
    backup_size / 1024 / 1024 AS [Size (MB)],
    'Log' AS [Backup Type]
FROM msdb.dbo.backupset
WHERE database_name = 'SampleDB'
  AND type = 'L'
ORDER BY backup_start_date DESC;

BACKUP LOG SampleDB
TO DISK = N'C:\Backups\SampleDB_Log_Recovery_1.trn'
WITH INIT, COMPRESSION;

WAITFOR DELAY '00:00:03';
INSERT INTO dbo.AnalyticsData (EventDate, EventType, EventData) VALUES

DELETE FROM dbo.AnalyticsData WHERE EventType = 'Recovery Test';

INSERT INTO #RecoveryTimestamps VALUES ('After delete (data loss)', GETDATE());

WITH INIT, COMPRESSION;
